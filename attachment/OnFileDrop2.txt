OnFileDrop2
{
	Filename = reference0
	_file = TOLOWER(Filename)
	//---- 壁紙変更
	if ".png" _in_ _file || ".jpg" _in_ _file || ".jpeg" _in_ _file ||".bmp" _in_ _file
	{
		ChoiceWallchange
	}
	//---- 音楽再生
	elseif ".mid" _in_ _file || ".mp3" _in_ _file || ".wav" _in_ _file || ".wma" _in_ _file
	{
		FUNCTIONEX("mciaudior.dll","stop")
		FUNCTIONEX("mciaudior.dll", "load", "%Filename")
		"\0\s[40]\![raise,OnPlayMusic]音楽ファイル再生中です。\w5\n演奏を止めたいときは、%keronameを\nダブルクリックしてください。\e"
	}
}
